7:export function CartProvider({ children }) {

9:  
10:  // French comments: Gérer le cart côté PrestaShop uniquement si connecté
11:  const [cartItems, setCartItems] = useState([])
12:  const [cartId, setCartId] = useState(null)
13:
14:  useEffect(() => {
15:    if (!customer) {
16:      setCartItems([])
17:      setCartId(null)
18:      return
19:    }

23:      try {
24:        const res = await fetch(`${API_URL}/carts?ws_key=${WS_KEY}&filter[id_customer]=${customer.id}&sort=[id_ASC]&display=full`)
25:        if (res.ok) {
26:          const xmlText = await res.text()
27:          const parser = new DOMParser()
28:          const xml = parser.parseFromString(xmlText, 'text/xml')
29:          const carts = xml.querySelectorAll('cart')
30:          
31:          if (carts.length > 0) {
32:            // French comments: Prend le dernier panier
33:            const lastCart = carts[carts.length - 1]
34:            const id = lastCart.querySelector('id')?.textContent
35:            setCartId(id)
36:
37:            // French comments: On pourrait récupérer les produits ici pour avoir les infos complètes.
38:            // Simplification: On stocke juste l'identifiant pour updateCart.
39:          }
40:        }
41:      } catch (err) {
42:        console.error('Erreur récupération panier:', err)
43:      }
44:    }

47:  }, [customer])

49:  // French comments: Fonction utilitaire pour synchroniser le panier avec PrestaShop
50:  async function syncCart(newItems) {
51:    if (!customer) return false
52:
53:    try {
54:      const rowsXml = newItems.map(item => `
55:        <cart_row>
56:          <id_product>${item.id}</id_product>
57:          <id_product_attribute>0</id_product_attribute>
58:          <id_address_delivery>0</id_address_delivery>
59:          <quantity>${item.quantity}</quantity>
60:        </cart_row>
61:      `).join('')
62:
63:      const cartXml = `<?xml version="1.0" encoding="UTF-8"?>
64:<prestashop>
65:  <cart>
66:    <id_customer>${customer.id}</id_customer>
67:    <id_currency>1</id_currency>
68:    <id_lang>1</id_lang>
69:    <associations>
70:      <cart_rows>${rowsXml}</cart_rows>
71:    </associations>
72:  </cart>
73:</prestashop>`
74:
75:      const method = cartId ? 'PUT' : 'POST'
76:      const cartXmlFinal = cartId 
77:        ? cartXml.replace('<cart>', `<cart><id>${cartId}</id>`)
78:        : cartXml
79:
80:      const res = await fetch(`${API_URL}/carts${cartId ? `?id=${cartId}&` : '?'}ws_key=${WS_KEY}`, {
81:         method,
82:         headers: { 'Content-Type': 'text/xml' },
83:         body: cartXmlFinal
84:      })
85:
86:      if (res.ok) {
87:         if (!cartId) {
88:            const resXmlText = await res.text()
89:            const xml = new DOMParser().parseFromString(resXmlText, 'text/xml')
90:            const newId = xml.querySelector('cart > id')?.textContent
91:            if (newId) setCartId(newId)
92:         }
93:         return true
94:      }
95:      return false
96:    } catch (err) {
97:      console.error('Erreur lors de la synchro panier:', err)
98:      return false
99:    }
100:  }
101:
102:  function addToCart(product, quantity = 1) {
103:    if (!customer) {
104:        alert("Veuillez vous connecter pour ajouter au panier.")
105:        return
106:    }
107:
108:    setCartItems(prev => {
109:      const existing = prev.find(i => i.id === product.id)
110:      let newItems
111:      if (existing) {
112:        newItems = prev.map(i =>
113:          i.id === product.id ? { ...i, quantity: i.quantity + quantity } : i
114:        )
115:      } else {
116:        newItems = [...prev, { ...product, quantity }]
117:      }
118:      syncCart(newItems)
119:      return newItems
120:    })
121:  }
122:
123:  function removeFromCart(id) {
124:    if (!customer) return
125:    setCartItems(prev => {
126:       const newItems = prev.filter(i => i.id !== id)
127:       syncCart(newItems)
128:       return newItems
129:    })
130:  }
131:
132:  function updateQuantity(id, quantity) {
133:    if (!customer) return
134:    if (quantity <= 0) return removeFromCart(id)
135:    setCartItems(prev => {
136:      const newItems = prev.map(i => i.id === id ? { ...i, quantity } : i)
137:      syncCart(newItems)
138:      return newItems
139:    })
140:  }
141:
142:  function clearCart() { 
143:      setCartItems([])
144:      setCartId(null)
145:  }
146:
147:  const total = cartItems.reduce((s, i) => s + parseFloat(i.price) * i.quantity, 0)
148:  const count = cartItems.reduce((s, i) => s + i.quantity, 0)
149:
150:  return (
151:    <CartContext.Provider value={{ cart: cartItems, cartId, addToCart, removeFromCart, updateQuantity, clearCart, total, count }}>
152:      {children}
153:    </CartContext.Provider>
154:  )
155:}
